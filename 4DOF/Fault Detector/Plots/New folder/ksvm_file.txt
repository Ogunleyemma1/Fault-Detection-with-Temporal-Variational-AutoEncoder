# validation_vae.py

import torch
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import json
import joblib
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVC
from sklearn.metrics import (
    roc_auc_score, average_precision_score, f1_score,
    precision_recall_fscore_support, confusion_matrix
)
from scipy.stats import skew, kurtosis
import seaborn as sns

# Parameters
SEQ_LEN = 100
NORMAL_FILE = "vae_input_data.csv"
STRUCT_FAULT_FILES = ["structural_fault_k2_reduced.csv", "structural_fault_c1_reduced.csv"]
SENSOR_FAULT_FILES = ["sensor_fault_x1_zero.csv", "sensor_fault_v3_noisy.csv"]
NORMAL_FRAC = 0.3
NORMAL_OFFSET = 0.4
FAULT_FRAC = 0.3

def extract_features(windows):
    feat_list = []
    for win in windows:
        feats = []
        for i in range(win.shape[1]):
            x = win[:, i]
            feats.append(np.nanmean(x))
            feats.append(np.nanstd(x))
            feats.append(np.nanmin(x))
            feats.append(np.nanmax(x))
            feats.append(np.nan_to_num(skew(x), nan=0.0, posinf=0.0, neginf=0.0))
            feats.append(np.nan_to_num(kurtosis(x), nan=0.0, posinf=0.0, neginf=0.0))
        feat_list.append(feats)
    return np.array(feat_list)

def preprocess(file, frac, offset=0.0, seq_len=SEQ_LEN):
    df = pd.read_csv(file)
    start, end = int(offset * len(df)), int((offset + frac) * len(df))
    data = df.iloc[start:end].values.astype(np.float32)
    return np.stack([data[i:i+seq_len] for i in range(len(data) - seq_len + 1)])

def load_vae_loss(model_path, mean_path, std_path, windows):
    from temporal_vae import VAE
    MODEL_PARAMS = dict(input_dim=12, latent_dim=16, hidden_dim=128, num_layers=2, dropout=0.3)
    mean, std = np.load(mean_path), np.load(std_path)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = VAE(**MODEL_PARAMS).to(device)
    model.load_state_dict(torch.load(model_path, map_location=device))
    model.eval()
    losses = []
    with torch.no_grad():
        for win in windows:
            x = torch.tensor(win[np.newaxis], dtype=torch.float32).to(device)
            recon = model(x)[0].cpu().numpy().squeeze()
            mse = np.mean((win - recon) ** 2)
            losses.append(mse)
    return np.array(losses)[:, None]

def load_datasets_and_features():
    X_norm = preprocess(NORMAL_FILE, frac=NORMAL_FRAC, offset=NORMAL_OFFSET)
    X_fault = np.concatenate([preprocess(f, frac=FAULT_FRAC) for f in STRUCT_FAULT_FILES + SENSOR_FAULT_FILES], axis=0)
    F_norm = extract_features(X_norm)
    F_fault = extract_features(X_fault)
    vae_loss_norm = load_vae_loss("temporal_vae_model.pt", "vae_mean.npy", "vae_std.npy", X_norm)
    vae_loss_fault = load_vae_loss("temporal_vae_model.pt", "vae_mean.npy", "vae_std.npy", X_fault)
    F_norm = np.hstack([F_norm, vae_loss_norm])
    F_fault = np.hstack([F_fault, vae_loss_fault])
    y_norm = np.zeros(F_norm.shape[0])
    y_fault = np.ones(F_fault.shape[0])
    X = np.vstack([F_norm, F_fault])
    y = np.concatenate([y_norm, y_fault])
    X = np.nan_to_num(X, nan=0.0, posinf=0.0, neginf=0.0)
    return X, y

def main():
    X, y = load_datasets_and_features()
    idx = np.random.permutation(len(X))
    split = int(0.7 * len(X))
    X_train, X_test = X[idx[:split]], X[idx[split:]]
    y_train, y_test = y[idx[:split]], y[idx[split:]]
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_test_scaled = scaler.transform(X_test)
    clf = SVC(kernel='rbf', C=1.0, probability=True, random_state=42)
    clf.fit(X_train_scaled, y_train)
    scores = clf.decision_function(X_test_scaled)
    thresholds = np.linspace(scores.min(), scores.max(), 100)
    best_f1, best_thresh = 0, 0
    for t in thresholds:
        preds = (scores > t).astype(int)
        f1 = f1_score(y_test, preds)
        if f1 > best_f1:
            best_f1, best_thresh = f1, t
    auroc = roc_auc_score(y_test, scores)
    auprc = average_precision_score(y_test, scores)
    print(f"\n--- SVM Results (Stat+VAE features) ---")
    print(f"AUROC: {auroc:.3f}   AUPRC: {auprc:.3f}")
    print(f"Best threshold: {best_thresh:.3f}  F1: {best_f1:.3f}")
    preds = (scores > best_thresh).astype(int)
    p, r, f1, _ = precision_recall_fscore_support(y_test, preds, average='binary', zero_division=0)
    print(f"Precision: {p:.3f}, Recall: {r:.3f}, F1: {f1:.3f}")
    cm = confusion_matrix(y_test, preds)
    plt.figure(figsize=(5,4))
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', xticklabels=["Normal","Fault"], yticklabels=["Normal","Fault"])
    plt.xlabel("Predicted"); plt.ylabel("True"); plt.title("Confusion Matrix (SVM)")
    plt.tight_layout(); plt.show()
    # Save everything!
    joblib.dump(scaler, "svm_scaler.pkl")
    joblib.dump(clf, "svm_model.pkl")
    with open("svm_best_threshold.json", "w") as f:
        json.dump({"svm_best_threshold": float(best_thresh)}, f)
    print("Saved scaler, SVM model, and threshold for future inference.")

if __name__ == "__main__":
    main()
